\section{Algorithms}
% =========================================================

\snippetlabel{Full algorithm — Doolittle LU decomposition (French keywords)}

\begin{algorithm}[H]
    \caption{Doolittle LU Decomposition}
    \label{alg:doolittle}
    \begin{algorithmic}[1]
        \Require Square matrix $A$ of size $n \times n$.

        \Function{Doolittle}{$A: \text{Matrix}, n: \text{Integer}$}
        \VarNoIndent{}
            \State $s, i, j, k, p$ : Integers
        \DebutNoIndent
            \For{$p \text{ from } 1 \text{ to } n$}
                \For{$j \text{ from } p \text{ to } n$}
                    \State $s \gets 0$
                    \For{$k \text{ from } 1 \text{ to } p-1$}
                        \State $s \gets s + A[p,k] \times A[k,j]$
                    \EndFor
                    \State $A[p,j] \gets A[p,j] - s$
                \EndFor
                \For{$i \text{ from } p+1 \text{ to } n$}
                    \State $s \gets 0$
                    \For{$k \text{ from } 1 \text{ to } p-1$}
                        \State $s \gets s + A[i,k] \times A[k,p]$
                    \EndFor
                    \State $A[i,p] \gets (1 \div A[p,p]) \times (A[i,p] - s)$
                \EndFor
            \EndFor
            \State \Return $A$
        \EndFunction
    \end{algorithmic}
\end{algorithm}
